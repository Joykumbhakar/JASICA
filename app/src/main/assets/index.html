Important Notice

We use Pins 10 & 11 for Bluetooth so you can upload code without unplugging the module!

Setup Instructions

1 Connect Hardware

Wire the HC-05 and Relay module to the Arduino exactly as shown in the diagram above. Keep everything powered off while wiring.

2 Upload Sketch

Copy or download the sketch below. Open the Arduino IDE on your computer, paste the code, and click Upload.

3 Pair Bluetooth

Open your Android Settings > Bluetooth. Find HC-05 and pair it. The default PIN is usually 1234 or 0000.

4 Connect in App

Open the Hey Robo app, go to the Devices tab, tap Scan, and add your HC-05. Tap Connect to start controlling!

Arduino Sketch (sketch.ino)
// Hey Robo - Smart Home Controller Sketch // Designed for iOS-Style Android App Integration #include <SoftwareSerial.h> // We use SoftwareSerial so hardware serial (Pin 0/1) // remains free for PC code uploading. SoftwareSerial bt(10, 11); // RX to BT TXD, TX to BT RXD // Pin Mapping to match App const int LEFT_LIGHT = 2; const int RIGHT_LIGHT = 3; const int LEFT_FAN = 4; const int RIGHT_FAN = 5; // Most relay modules are Active-LOW (LOW = ON). // Change these if your relays act in reverse. #define RELAY_ON LOW #define RELAY_OFF HIGH String command = ""; void setup() { pinMode(LEFT_LIGHT, OUTPUT); pinMode(RIGHT_LIGHT, OUTPUT); pinMode(LEFT_FAN, OUTPUT); pinMode(RIGHT_FAN, OUTPUT); // Ensure everything is OFF initially digitalWrite(LEFT_LIGHT, RELAY_OFF); digitalWrite(RIGHT_LIGHT, RELAY_OFF); digitalWrite(LEFT_FAN, RELAY_OFF); digitalWrite(RIGHT_FAN, RELAY_OFF); Serial.begin(9600); // For PC Serial Monitor bt.begin(9600); // Default HC-05 baud rate Serial.println("Hey Robo System Ready."); } void loop() { if (bt.available()) { char c = bt.read(); if (c == '\n') { command.trim(); processCommand(command); command = ""; // Reset buffer } else { command += c; } } } void processCommand(String cmd) { Serial.println("App Sent: " + cmd); // Lights if (cmd.equalsIgnoreCase("turn on left light")) digitalWrite(LEFT_LIGHT, RELAY_ON); else if (cmd.equalsIgnoreCase("turn off left light")) digitalWrite(LEFT_LIGHT, RELAY_OFF); else if (cmd.equalsIgnoreCase("turn on right light")) digitalWrite(RIGHT_LIGHT, RELAY_ON); else if (cmd.equalsIgnoreCase("turn off right light")) digitalWrite(RIGHT_LIGHT, RELAY_OFF); // Fans else if (cmd.equalsIgnoreCase("turn on left fan")) digitalWrite(LEFT_FAN, RELAY_ON); else if (cmd.equalsIgnoreCase("turn off left fan")) digitalWrite(LEFT_FAN, RELAY_OFF); else if (cmd.equalsIgnoreCase("turn on right fan")) digitalWrite(RIGHT_FAN, RELAY_ON); else if (cmd.equalsIgnoreCase("turn off right fan")) digitalWrite(RIGHT_FAN, RELAY_OFF); }
